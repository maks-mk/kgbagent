# ===========================
# 4) Лимиты и Безопасность
# ===========================
# Максимум одновременно выполняемых tool calls в одном batch (целое >= 1).
# 1 отключает параллельность; внутренние подзапросы инструментов не учитываются.
MAX_PARALLEL_TOOL_CALLS=4

# Финальный лимит вывода инструмента для контекста модели (символов).
MAX_TOOL_OUTPUT=15000

# Лимит вывода, собираемого до сжатия или финального сокращения.
# Эффективное значение не может быть меньше MAX_TOOL_OUTPUT.
MAX_RAW_TOOL_OUTPUT=100000

# Семантически сжимать слишком большой вывод shell, web-, listing- и MCP-инструментов
# через Headroom ContentRouter: тип содержимого (лог, поисковая выдача, JSON-массив,
# табличные данные, текст) определяется автоматически, отдельных параметров в mcp.json
# не нужно. Файловые инструменты никогда не сжимаются.
# Логи, которые роутер сжал недостаточно, дополнительно проходят через LogCompressor
# с лимитами, посчитанными от MAX_TOOL_OUTPUT: чем больше бюджет, тем больше ошибок,
# предупреждений и стеков сохраняется. Результат, доносящий меньше диагностики, чем
# структурное сокращение, отклоняется.
# При отключении или недоступности Headroom применяется структурное сокращение.
ENABLE_HEADROOM_COMPRESSION=false

# Максимальный размер контента одного поискового подзапроса (символов).
# Обычно больше, чем MAX_TOOL_OUTPUT, чтобы захватывать длинные статьи.
MAX_SEARCH_CHARS=20000

# Максимальный размер файла для чтения/записи.
# Число без суффикса трактуется как байты.
# Для удобства лучше указывать явные единицы: 4MB, 300MiB, 1GiB.
MAX_FILE_SIZE=300MiB

# Максимальное количество строк при чтении файла (защита от чтения логов на 1млн строк).

# Максимальное количество фоновых процессов (если ENABLE_PROCESS_TOOLS=true).
MAX_BACKGROUND_PROCESSES=5

# ===========================
# 5) Память и Сеть
# ===========================
# Порог для estimate_tokens(messages), после которого агент начинает рассматривать
# суммаризацию истории.
# Как считается estimate_tokens:
# - если доступен tiktoken: используется cl100k_base + небольшой overhead на сообщение;
# - если tiktoken недоступен: fallback-оценка len(text)//3 по содержимому сообщений и tool_calls.
# Это внутренний эвристический счётчик агента, а не точный provider-side token usage из статус-бара.
# К estimate_tokens добавляется SUMMARY_RESERVED_TOKENS: запас на системные инструкции,
# tool schemas и служебную provider-обвязку, которые провайдер считает в input tokens,
# но которых нет в сохранённой истории сообщений. Поставьте 0, если нужен только
# "чистый" счётчик истории.
# Дополнительно есть "мягкая зона": чуть выше порога суммаризация может быть отложена,
# если сжимать пока почти нечего или в истории слишком мало human-turns.
SESSION_SIZE=8000
SUMMARY_RESERVED_TOKENS=3000

# Сколько последних сообщений оставлять "сырыми" (без сжатия) при суммаризации.
# Для рабочих чатов обычно хватает 4-6.
SUMMARY_KEEP_LAST=4

# Настройки повторных попыток для сетевых запросов (Search, Fetch).
MAX_RETRIES=3
RETRY_DELAY=2
